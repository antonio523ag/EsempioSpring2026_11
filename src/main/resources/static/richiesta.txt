
cliente{
    id bigint primary key autoincrement
    nome varchar(255) not null
    cognome varchar(255) not null
}

domicili_cliente{
id_cliente bigint foreignkey -> cliente(id)
id_domicilio bigint foreingKey -> domicilio(id)
}

domicilio{
    id bigint primarykey autoincrement
    via varchar(255) not null
    citta varchar(255) not null
    cap varchar(255) not null
    civico varchar(255)
}

fatturazione{
id bigint primaryKey autoincrement
codice varchar(255) not null
dataEmissione date notNull
importo double notnull
dataScadenza date notnull
tipologia String notnull
id_domicilio bigint foreignkey domicilio(id)
}

pagamento{
id_fatturazione foreignkey fatturazione(id)
id_cliente foreignkey cliente(id)
dataPagamento date notnull
}


endpoint che trovi gli utenti in base a uno o più dei seguenti parametri
{
fatture pagate(true o false)
città
cognomeEONome
codiceFattura
tipologiaFattura
}
l'endpoint sarà paginato, con il numero pagina da pathvariabile e il numero di elementi fisso a 50
